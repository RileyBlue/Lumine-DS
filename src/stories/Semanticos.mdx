export const semantic = tokens.semantic

export const find = (path) => path.split('.').reduce((node, part) => node[part], tokens)

export const isAlias = (value) => typeof value === 'string' && value.startsWith('{')

export const resolve = (value) => {
  if (isAlias(value)) return resolve(find(value.slice(1, -1)).$value)
  if (typeof value === 'object' && value !== null && !('unit' in value) && !('colorSpace' in value)) {
    return Object.fromEntries(Object.entries(value).map(([key, item]) => [key, resolve(item)]))
  }
  return value
}

export const target = (alias) => alias.slice(1, -1)

export const flatten = (group, path) =>
  Object.entries(group)
    .filter(([name]) => !name.startsWith('$'))
    .flatMap(([name, node]) =>
      node.$value === undefined
        ? flatten(node, `${path}.${name}`)
        : [{ name: `${path}.${name}`, alias: node.$value, value: resolve(node.$value) }],
    )

export const size = ({ value, unit }) => `${value}${unit}`

export const color = ({ colorSpace, components, alpha = 1 }) =>
  colorSpace === 'oklch'
    ? `oklch(${components.join(' ')} / ${alpha})`
    : `color(${colorSpace} ${components.join(' ')} / ${alpha})`

export const paint = (name) => color(resolve(find(`semantic.color.${name}`).$value))

export const sans = tokens.primitive['font-family'].sans.$value
  .map((name) => (name.includes(' ') ? `"${name}"` : name))
  .join(', ')

export const line = `1px solid ${paint('border.neutral.default')}`

export const accent = paint('background.brand.default')

export const Label = ({ name, children }) => (
  <div style={{ fontSize: '0.75rem', lineHeight: '1.25rem', overflowWrap: 'anywhere' }}>
    <code style={{ fontWeight: 600 }}>{name.replace('semantic.', '').replaceAll('.', '.\u200B')}</code>
    <div style={{ color: paint('text.neutral.subtle.default') }}>{children}</div>
  </div>
)

export const Arrow = ({ alias }) => <div>→ {target(alias).replace('primitive.', '')}</div>

export const Grid = ({ children }) => (
  <div
    style={{
      display: 'grid',
      gridTemplateColumns: 'repeat(auto-fill, minmax(14rem, 1fr))',
      gap: '1.25rem',
      padding: '0.5rem 0 1rem',
    }}
  >
    {children}
  </div>
)

export const Box = ({ background, style, children }) => (
  <div
    style={{
      display: 'flex',
      alignItems: 'center',
      justifyContent: 'center',
      gap: '0.5rem',
      height: '5rem',
      borderRadius: '0.5rem',
      border: line,
      background: paint(background),
      fontFamily: sans,
      marginBottom: '0.25rem',
      ...style,
    }}
  >
    {children}
  </div>
)

export const colorTokens = (element) => flatten(semantic.color[element], `semantic.color.${element}`)

export const short = (name) => name.replace('semantic.color.', '')

export const textOn = (name) => {
  const [, role, ...rest] = name.split('.')
  if (rest.includes('disabled')) return 'text.neutral.disabled'
  if (role === 'neutral') {
    return rest.includes('inverted') ? 'text.neutral.inverted.default' : 'text.neutral.default'
  }
  return rest.includes('subtle') ? `text.${role}.default` : `text.${role}.inverted.default`
}

export const backgroundFor = (name) => {
  const [, role, ...rest] = name.split('.')
  if (rest.includes('disabled')) return ['background.neutral.disabled']
  if (rest.includes('inverted')) {
    return [role === 'neutral' ? 'background.neutral.inverted.default' : `background.${role}.default`]
  }
  if (role === 'neutral') return ['background.neutral.default']
  return ['background.neutral.default', `background.${role}.subtle.default`]
}

export const Backgrounds = () => (
  <Unstyled>
    <Grid>
      {colorTokens('background').map(({ name, alias }) => (
        <div key={name}>
          <Box background={short(name)} style={{ color: paint(textOn(short(name))) }}>
            Clase de ballet
          </Box>
          <Label name={name}>
            <Arrow alias={alias} />
            <div>con {textOn(short(name))}</div>
          </Label>
        </div>
      ))}
    </Grid>
  </Unstyled>
)

export const Foreground = ({ element }) => (
  <Unstyled>
    <Grid>
      {colorTokens(element).map(({ name, alias, value }) => (
        <div key={name}>
          <div style={{ display: 'flex', gap: '0.5rem' }}>
            {backgroundFor(short(name)).map((background) => (
              <div key={background} style={{ flex: 1 }}>
                <Box background={background} style={{ color: color(value) }}>
                  {element === 'icon' ? <IconBell size={28} aria-hidden="true" /> : 'Clase de ballet'}
                </Box>
              </div>
            ))}
          </div>
          <Label name={name}>
            <Arrow alias={alias} />
            <div>sobre {backgroundFor(short(name)).join(' y ')}</div>
          </Label>
        </div>
      ))}
    </Grid>
  </Unstyled>
)

export const Borders = () => (
  <Unstyled>
    <Grid>
      {colorTokens('border').map(({ name, alias, value }) => {
        const isFocus = name.endsWith('focus')
        const width = size(resolve(semantic['border-width'][isFocus ? 'focus' : 'default'].$value))
        return (
          <div key={name}>
            <Box
              background="background.neutral.default"
              style={
                isFocus
                  ? { outline: `${width} solid ${color(value)}`, outlineOffset: '2px' }
                  : { border: `${width} solid ${color(value)}` }
              }
            />
            <Label name={name}>
              <Arrow alias={alias} />
              <div>sobre background.neutral.default</div>
            </Label>
          </div>
        )
      })}
    </Grid>
  </Unstyled>
)

# Semánticos

Son los tokens que dicen para qué se usa cada valor. Cada uno apunta a un primitivo: aquí se ve su
nombre y, después de la flecha, el primitivo al que apunta. Para no repetir, los nombres van sin el
prefijo `semantic.` y los destinos sin `primitive.`.

Esta página lee `tokens/tokens.json`. Los componentes todavía no usan estos tokens.
